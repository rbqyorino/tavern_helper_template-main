rule:
  - 在下次回复的末尾，输出<PhoneUpdateVariable>块.
  - Use `_.set('path', 'old', 'new')` for direct assignment or state changes.
  - Use `_.insert('path', 'key', 'value')` to add a property to an object.
  - Use `_.insert('path', 'item')` to add an element to an array.
  - Use `_.remove('path', 'item_or_key')` to remove an element from an array or object.
 
example: |-
  _.set('手机数据.当前时间', '2025-01-23 18:00:00', '2025-01-23 19:00:00'); // 时间推进到一小时后
  _.insert('手机数据.用户.空间动态', '2025-01-23 14:30:00', {内容: '今天天气真不错', 评论:{$meta: {extensible: true}}}); // <user>发布新动态
  _.insert('手机数据.用户.空间动态.2025-01-23 14:30:00.评论', '1', {ID: '柳木诗梦', 发言内容: '确实呢，要不要一起出去走走？'}); // 诗梦评论
  _.remove('手机数据.用户.空间动态', '2025-01-23 14:30:00'); // <user>删除了自己之前发布的动态
  _.remove('手机数据.联系人.柳木诗梦.空间动态.2025-01-23 13:00:00.评论', '2'); // 删除诗梦动态下的某条评论
  _.remove('手机数据.联系人.柳木诗梦.聊天记录', '2025-01-23 18:59:00'); // <user>撤回了之前发送的消息

    </PhoneAnalysis>
